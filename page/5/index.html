解题思路

class Solution {

    /**
     * @param Integer[] $nums
     * @return Integer
     */
    function removeDuplicates(&$nums) {
        $size = sizeof($nums);// 防止数组越界
        foreach($nums as $key => $item){
            if($key + 1 <= $size) {
                if($nums[$key] === $nums[$key + 1]){
                    unset($nums[$key]);
                }
            }
        }
        return count($nums);
    }
}
